    src/serial_transport.cpp src/settings.cpp src/udp_probe.cpp src/controller_client.cpp src/waypoints.cpp src/streamer.cpp src/probe_motion.cpp
    ../common/protocol/mach3_usb.c ../common/protocol/mach3_probe.c ../common/protocol/mach3_control.c ../common/motion/controller.c)
target_include_directories(sn-mach3-core PUBLIC src ../common)

if(SN_BUILD_MACH3_PLUGIN)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "Mach3 DLL requires Windows, MSVC with MFC and a Win32 generator (-A Win32)")
    endif()
    set(MACH3_SDK_DIR "" CACHE PATH "Extracted SDK directory containing BlankMovement and MachIncludes")
    foreach(file BlankMovement/MachDevice.cpp BlankMovement/XMLProfile.cpp MachIncludes/Engine.h)
        if(NOT EXISTS "${MACH3_SDK_DIR}/${file}")
            message(FATAL_ERROR "Missing SDK file ${file}; set MACH3_SDK_DIR to the extracted SDK directory")
        endif()
    endforeach()
    set(MACH3_ABI_DIR "" CACHE PATH "MachIncludes directory extracted from the official Mach3 Plugin Wizard 1.1a")
    if(NOT EXISTS "${MACH3_ABI_DIR}/_TrajectoryControl.h")
        message(FATAL_ERROR "Set MACH3_ABI_DIR to the official Plugin Wizard MachIncludes directory; modern MFC class layouts are incompatible with Mach3")
    endif()
    enable_language(RC)
    set(CMAKE_MFC_FLAG 2)
    # Stage the official wrapper so its quoted includes use our stdafx.h.
    set(stage "${CMAKE_CURRENT_BINARY_DIR}/sdk-wrapper")
    file(MAKE_DIRECTORY "${stage}")
    foreach(file MachDevice.cpp MachDevice.h XMLProfile.cpp XMLProfile.h)
        configure_file("${MACH3_SDK_DIR}/BlankMovement/${file}" "${stage}/${file}" COPYONLY)
    endforeach()
    # The old static-MFC sample omitted this switch in SetProName. With shared
    # modern MFC, Mach3 has no CWinApp in our runtime until we select DLL state.
    file(READ "${stage}/MachDevice.cpp" mach_device)
    string(REPLACE "static CString strPlgName, strVer;"
        "AFX_MANAGE_STATE(AfxGetStaticModuleState());\n      static CString strPlgName, strVer;"
        mach_device "${mach_device}")
    # An external controller must not request Mach3's parallel-port engine.
    string(REGEX REPLACE "return true; //start the printer port[^\r\n]*"
        "return false; // StepperNinja supplies external motion; no printer port."
        mach_device "${mach_device}")
    string(REPLACE "#include \"TrajectoryControl.h\"" "#include \"mach3_abi.h\"" mach_device "${mach_device}")
    string(REPLACE "#include \"Mach4View.h\"" "" mach_device "${mach_device}")
    file(WRITE "${stage}/MachDevice.cpp" "${mach_device}")
    # The official Wizard uses plain structs to preserve Mach3's VC7.1 ABI.
    # Never compile its shared blocks as current-MFC CWnd/CView subclasses.
    file(COPY "${MACH3_ABI_DIR}/" DESTINATION "${stage}/MachIncludes")
    add_library(StepperNinja SHARED "${stage}/MachDevice.cpp" "${stage}/XMLProfile.cpp"
        src/MachDevImplementation.cpp config/config.rc config/StepperNinja.def)
    target_include_directories(StepperNinja PRIVATE src config "${stage}" "${stage}/MachIncludes")
    target_compile_definitions(StepperNinja PRIVATE _AFXDLL _MBCS _USRDLL WINVER=0x0601 _WIN32_WINNT=0x0601)
    target_link_libraries(StepperNinja PRIVATE sn-mach3-core ole32 oleaut32 uuid)
    set_target_properties(StepperNinja PROPERTIES PREFIX "")
endif()
